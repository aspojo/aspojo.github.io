Linux 日志监控脚本使用说明（简洁版）

概述

monitor_logs.sh 监控 /var/local/bimss/logs/ 子文件夹中的滚动日志（.log.[数字]），压缩成 .tar.gz 归档到 /var/local/bimss/logs/archive/YYYY-MM-DD/，删除原文件，并清理指定天数前目录（默认7天）。

依赖

  • Linux (bash, find, tar, date)。
  • 写权限到 /var/local/bimss/logs/。

安装

  1. 创建目录：sudo mkdir -p /var/local/bimss/logs/archive
  2. 保存脚本到 /usr/local/bin/monitor_logs.sh（见下）。
  3. 权限：sudo chmod +x /usr/local/bin/monitor_logs.sh

脚本内容

#!/bin/bash

LOGS_DIR="/var/local/bimss/logs"
ARCHIVE_DIR="/var/local/bimss/logs/archive"
DAYS_TO_KEEP=${1:-7}

mkdir -p "${ARCHIVE_DIR}"

get_rolling_logs() {
    find "${LOGS_DIR}" -not -path "${LOGS_DIR}/archive/*" -type f -name "*.log.[0-9]*" -printf '%p\n' | sort
}

archive_file() {
    local log_file="$1"
    local subdir=$(basename "$(dirname "${log_file}")")
    local timestamp=$(date '+%Y%m%d_%H%M%S')
    local file_basename=$(basename "${log_file}")
    local date_dir=$(date '+%Y-%m-%d')
    local target_subdir="${ARCHIVE_DIR}/${date_dir}"
    mkdir -p "${target_subdir}"
    local archive_name="${subdir}_${file_basename}_${timestamp}.tar.gz"
    local archive_path="${target_subdir}/${archive_name}"
    tar -czf "${archive_path}" -C "$(dirname "${log_file}")" "${file_basename}"
    if [[ $? -eq 0 ]]; then
        echo "已压缩并归档：${log_file} -> ${archive_path}"
        rm -f "${log_file}"
        echo "已删除原文件：${log_file}"
    else
        echo "压缩失败：${log_file}"
    fi
}

cleanup_old_archives() {
    echo "开始清理 ${DAYS_TO_KEEP} 天前归档..."
    local current_date=$(date '+%Y-%m-%d')
    local cutoff_date=$(date -d "${current_date} - ${DAYS_TO_KEEP} days" '+%Y-%m-%d')
    echo "当前: ${current_date}，截止: ${cutoff_date}"
    local deleted_count=0
    while IFS= read -r date_dir; do
        if [[ -n "${date_dir}" && -d "${date_dir}" ]]; then
            local dir_date=$(basename "${date_dir}")
            if [[ "${dir_date}" < "${cutoff_date}" ]]; then
                rm -rf "${date_dir}"
                if [[ $? -eq 0 ]]; then
                    echo "已删除: ${dir_date}"
                    deleted_count=$((deleted_count + 1))
                fi
            fi
        fi
    done < <(find "${ARCHIVE_DIR}" -maxdepth 1 -type d -name "20[0-9][0-9]-[0-9][0-9]-[0-9][0-9]" 2>/dev/null)
    [[ ${deleted_count} -eq 0 ]] && echo "无旧目录" || echo "删除 ${deleted_count} 个"
    find "${ARCHIVE_DIR}" -type d -empty -delete
    echo "清理空文件夹完成"
}

main() {
    local current_files=$(get_rolling_logs)
    while IFS= read -r file; do
        [[ -n "${file}" && -f "${file}" ]] && archive_file "${file}"
    done <<< "${current_files}"
    cleanup_old_archives
}

main

运行

  • 手动：./monitor_logs.sh [天数]（默认7）。
  • Cron（每5min）： sudo crontab -e，添加 */5 * * * * /usr/local/bin/monitor_logs.sh >> /var/log/monitor_logs.log 2>&1

测试

  1. 生成测试文件：用 test_logs.sh（之前提供）。
  2. 创建滚动日志：touch /var/local/bimss/logs/bzz/test.log.1
  3. 运行脚本，检查归档/删除。
  4. 清理测试：运行后，旧日期目录（如8天前）应删除。

注意

  • 备份前测试，避免误删。
  • 自定义：编辑 LOGS_DIR/ARCHIVE_DIR。
  • 问题：查权限/日志输出。
